34def stateID(url, seed): 

35 """Create hash of the time + server path + a seed makes an unique SID for each session. 

36 

37 :param url: The base URL for this site 

38 :return: The hex version of the digest 

39 """ 

40 # Seed may be bytes or unicode 

41 try: 

42 seed = seed.encode() 

43 except AttributeError: 

44 pass 

45 

46 # Mostly cargo cult, we could just use rndstr(16) 

47 ident = hashlib.new("sha256") 

48 ident.update(repr(time.time()).encode()) 

49 ident.update(url.encode()) 

50 ident.update(seed) 

51 # Mix in some randomness, as time.time() does not have enough 

52 # accuracy on all platforms to make this unique for identical 

53 # seeds and urls. 

54 # The accuracy can be pretty bad, around 20ms, unless one uses 

55 # one of the high precision timers (e.g. py3 timeit.default_timer()) 

56 # that use stuff like the performance counters in the CPU. 

57 # But even that can and does break on VMs, where the performance counters 

58 # are virtualized too, so better play safe than sorry here. 

59 ident.update(rndstr(8).encode()) 

60 return ident.hexdigest()
